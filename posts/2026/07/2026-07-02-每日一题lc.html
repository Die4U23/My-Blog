class Solution {
public:
    vector<vector<int>> verticalTraversal(TreeNode* root) {
        vector<tuple<int, int, int>> data;
        auto dfs = [&](this auto&& dfs, TreeNode* node, int row, int col) -> void{
            if(!node)return;
            data.emplace_back(col, row, node -> val);
            dfs(node -> left, row + 1, col - 1);
            dfs(node -> right, row + 1, col + 1);
        };
        dfs(root, 0, 0);
        vector<vector<int>> ans;
        ranges::sort(data);
        int last_col = INT_MIN;
        for(auto &[col, _, val]: data){
            if(col != last_col){
                last_col = col;
                ans.push_back({});
            }
            ans.back().push_back(val);
        }
        return ans;
    }
};

class Solution {
public:
    bool flipEquiv(TreeNode* root1, TreeNode* root2) {
        if(!root1 || !root2)return !root1 && !root2;
        bool ans = true;
        std::unordered_map<int, int> map;
        auto dfs = [&](this auto&& dfs, TreeNode* node, bool state, int parent) -> void{
            if(!node)return;
            int value = node -> val;
            if(state)map.emplace(value, parent);
            else{
                auto it = map.find(value);
                if(it != map.end()){
                    if(it -> second != parent){
                        ans = false;
                        return;
                    }
                    map.erase(value);
                }
                else{
                    ans = false;
                    return;
                }
            }
            dfs(node -> left, state, value);
            dfs(node -> right, state, value);
        };
        dfs(root1, true, -1);
        dfs(root2, false, -1);
        if(ans && !map.empty())return false;
        return ans;
    }
};

class Solution {
public:
    vector<int> findFrequentTreeSum(TreeNode* root) {
        unordered_map<int, int> map;
        int max = 0;
        auto dfs = [&](this auto&& dfs, TreeNode* node){
            if(node == nullptr)return 0;
            int left = dfs(node -> left);
            int right = dfs(node -> right);
            int sum = left + right + node -> val;

            if(map[sum] > max) max = map[sum];

        for(const auto& [sum, count]: map){
            if(count == max)ans.push_back(sum);
        }
        return ans;
    }

class Solution {
public:
    int maxAncestorDiff(TreeNode* root) {
        if(!root)return 0;
        int ans = 0;
        int temp = root -> val;
        auto dfs =[&](this auto&& dfs, TreeNode* node, int min, int max) -> void{
            if(!node)return;
            if(node -> val < min){
                min = node -> val;
                if(max - min > ans)ans = max - min;
            }
            else if(node -> val > max){
                max = node -> val;
                if(max - min > ans)ans = max - min;
            }
            dfs(node -> left, min, max);
            dfs(node -> right, min, max);
            return;
        };
        dfs(root, temp, temp);
        return ans;
    }
};

public:

        int ans = 0;
        auto dfs = [&](this auto&& dfs, TreeNode* node, bool state, int length) -> void{
            if(!node)return;

1080.根到叶路径的不足节点

1080.根到叶路径到不足节点

这题第一遍做想复杂了，使用哈希表记录路径值再进行处理，效率较低


class Solution {
public:
    int makeMap(TreeNode* node, unordered_map<TreeNode*, int>& map){
        if(!node)return INT_MIN;
        else if(!(node -> left) && !(node -> right))return map[node];
        else{
            map[node] = max(makeMap(node -> left, map), makeMap(node -> right, map));
            return map[node];
        }
    }
    TreeNode* sufficientSubset(TreeNode* root, int limit) {
        unordered_map<TreeNode*, int> map;
        auto dfs = [&](this auto&& dfs, TreeNode* node, int cur) -> void{
            if(!node)return;
            cur += node -> val;
            if(!(node -> left) && !(node -> right)){
                map[node] = cur;
            }
            dfs(node -> left, cur);
            dfs(node -> right, cur);
        };
        auto dfsDel = [&](this auto&& dfsDel, TreeNode* node) -> void{
            if(!node)return;
            if(node -> left && map[node -> left] < limit)node -> left = nullptr;
            if(node -> right && map[node -> right] < limit)node -> right = nullptr;
            dfsDel(node -> left);
            dfsDel(node -> right);
            return;
        };
        dfs(root, 0);
        makeMap(root, map);
        dfsDel(root);
        return map[root] < limit? nullptr: root;
    }
};

答案的算法使用了递归，用limit减去val，返回子树处理的结果，同时进行了探索和回溯


class Solution {
public:
    TreeNode* sufficientSubset(TreeNode* root, int limit) {
        if(root == nullptr) return root;
        if(root->left == nullptr && root->right == nullptr) 
        {
            return root->val < limit?  nullptr : root;
        }
        if(root->left) root->left = sufficientSubset(root->left, limit-root->val);
        if(root->right) root->right = sufficientSubset(root->right, limit-root->val);
        return root->left || root->right ? root : nullptr;
    }
};

class Solution {

        if(!root)return true;
        if(!isValidBST(root -> left))return false;
        if(root -> val <= pre)return false;
        pre = root -> val;
        if(!isValidBST(root -> right))return false;
        return true;
    }

530.二叉搜索树的最小绝对差

530.二叉搜索树的最小绝对差

中序遍历即为从小到大的顺序


class Solution {
public:
    int getMinimumDifference(TreeNode* root) {
        int ans = 100000000, pre = -10000000;
        auto mid = [&](this auto&& mid, TreeNode* node) -> void{
            if(!node)return;
            mid(node -> left);
            int val = node -> val;
            ans = min(ans, val - pre);
            pre = val;
            mid(node -> right);
            return;
        };
        mid(root);
        return ans;
    }
};

这个代码用lambda捕获ans和pre，使得内部不敢把这两个局部变量放进寄存器中，降低了效率。
将lambda的方式改成纯函数风格可以降低副作用。
采用尾调用形式可以让编译器消除栈帧
更为优秀的代码:


class Solution {
    std::pair<int, int> dfs(TreeNode* node, int pre, int ans) {
        if (!node) return {ans, pre};
        auto [left_ans, left_pre] = dfs(node->left, pre, ans);
        if (left_pre != -1) {
            int diff = node->val - left_pre;
            if (diff < left_ans) left_ans = diff;
        }
        return dfs(node->right, node->val, left_ans);
    }
public:
    int getMinimumDifference(TreeNode* root) {
        return dfs(root, -1, INT_MAX).first;
    }
};

此外可以用迭代替代递归，更符合编译器


class Solution {
public:
    int getMinimumDifference(TreeNode* root) {
        int ans = INT_MAX;
        int pre = -1;
        bool first = true;
        std::stack<TreeNode*> st;
        TreeNode* cur = root;
        while (cur || !st.empty()) {
            while (cur) {
                st.push(cur);
                cur = cur->left;
            }
            cur = st.top();
            st.pop();
            if (first) {
                first = false;
            } else {
                int diff = cur->val - pre;
                if (diff < ans) ans = diff;
            }
            pre = cur->val;
            cur = cur->right;
        }
        return ans;
    }
};

2476.二叉搜索树最近节点查询

2476.二叉搜索树最近节点查询

中序遍历得到有序队列然后二分获得上下节点


class Solution {
    vector<int> v;

    void dfs(TreeNode* node){
        if(!node)return;
        dfs(node -> left);
        v.push_back(node -> val);
        dfs(node -> right);
    };
public:
    vector<vector<int>> closestNodes(TreeNode* root, vector<int>& queries) {
      dfs(root);
      int s = v.size();
      vector<vector<int>> ans;
      for(int q: queries){
        int j = ranges::lower_bound(v, q) - v.begin();
        int mx = j < s ? v[j] : -1;
        if(j == s || v[j] != q)j--;
        int mn = j >= 0? v[j]: -1;
        ans.push_back({mn, mx});
      }  
      return ans;
    }
};

两种dfs的方法


class Solution {
    pair dfs(TreeNode* node){
        if(!node)return {0, nullptr};
        auto [lh, ln] = dfs(node -> left);
        auto [rh, rn] = dfs(node -> right);
        if(lh > rh)return{lh + 1, ln};
        if(lh < rh)return{rh + 1, rn};
        return {lh + 1, node};
    };
public:
    TreeNode* lcaDeepestLeaves(TreeNode* root) {
        return dfs(root).second;
    }
};
			

class Solution {
public:
    TreeNode* lcaDeepestLeaves(TreeNode* root) {
        int maxDepth = -1;
        TreeNode* ans = nullptr;
        auto dfs = [&](this auto&& dfs, TreeNode* node, int depth) -> int{
            if(!node){
                maxDepth = max(maxDepth, depth);
                return depth;
            }
            int l = dfs(node -> left, depth + 1);
            int r = dfs(node -> right, depth + 1);
            if( l == r && r== maxDepth)ans = node;
            return max(l, r);
        };
        dfs(root, 0);
        return ans;
    }
};
